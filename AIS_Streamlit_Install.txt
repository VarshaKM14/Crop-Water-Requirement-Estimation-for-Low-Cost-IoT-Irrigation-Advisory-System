@echo off
setlocal

echo === Step 1: Download Python 3.10.0 ===
if not exist python-3.10.0.exe (
    curl -L -o python-3.10.0.exe https://www.python.org/ftp/python/3.10.0/python-3.10.0-amd64.exe
)

echo === Step 2: Install Python 3.10.0 silently ===
python-3.10.0.exe /quiet InstallAllUsers=1 PrependPath=1 Include_test=0

echo === Step 3: Wait for install ===
timeout /t 10 >nul

echo === Step 4: Create virtual environment ===
py -3.10 -m venv venv310

echo === Step 5: Activate virtual environment ===
call venv310\Scripts\activate

echo === Step 6: Upgrade pip and tools ===
python -m ensurepip
python -m pip install --upgrade pip setuptools wheel

echo === Step 7: Install required packages ===
pip install streamlit matplotlib pandas scikit-learn seaborn joblib

echo === Setup Completed ===
pause
